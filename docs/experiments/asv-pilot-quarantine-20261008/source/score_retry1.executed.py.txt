"""Score one completed ASV pilot checkpoint on the predeclared dev media only."""

from __future__ import annotations

import argparse
import json
import subprocess
import sys
from dataclasses import asdict
from pathlib import Path

from deepfake_detection_archive.config import InferenceConfig
from deepfake_detection_archive.contracts import load_scores
from deepfake_detection_archive.data.manifest import file_sha256, load_manifest
from deepfake_detection_archive.evaluation.report import evaluate_records
from deepfake_detection_archive.tracking import atomic_json, utc_now


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--project-root", type=Path, required=True)
    parser.add_argument(
        "--run",
        choices=(
            "aasist-l-seed43-retry1",
        ),
        required=True,
    )
    args = parser.parse_args()
    root = args.project_root.resolve(strict=True)
    plan = root / "outputs/experiment-planning/asv-pilot-quarantine"
    experiment = root / "outputs/experiments/asv-pilot-quarantine" / args.run
    train = json.loads((experiment / "training/run.json").read_text())
    if train["status"] != "complete":
        raise ValueError("training must finish successfully before scoring")
    checkpoint = experiment / "training/best.pt"
    checkpoint_hash = file_sha256(checkpoint)
    if checkpoint_hash != train["checkpoint"]["sha256"]:
        raise ValueError("selected checkpoint hash mismatch")
    planned = json.loads((plan / f"{args.run}.json").read_text())
    dev_path = root / "data/asvspoof2019-la-pilot-quarantine/dev.jsonl"
    records = load_manifest(dev_path)
    if len(records) != 512 or any(r.split != "dev" for r in records):
        raise ValueError("requires 512 dev samples, no train/holdout/eval")
    output = experiment / "dev-inference"
    if output.exists():
        raise ValueError("inference output already exists; do not overwrite")
    config = InferenceConfig(
        manifest=str(dev_path),
        data_root=planned["data_root"],
        output_dir=str(output),
        modality="audio",
        checkpoint=str(checkpoint),
        checkpoint_sha256=checkpoint_hash,
        device="cuda",
        model_kind="audio_waveform",
        audio=planned["audio"],
        segments=planned["segments"],
        segment_samples=planned["segment_samples"],
        batch_size=planned["batch_size"],
        seed=planned["training"]["seed"],
    )
    config_path = experiment / "dev-inference-config.json"
    atomic_json(config_path, config.to_dict())
    command = [
        sys.executable,
        "-m",
        "deepfake_detection_archive",
        "infer",
        "--config",
        str(config_path),
    ]
    with (
        (experiment / "dev-inference.stdout").open("x") as stdout,
        (experiment / "dev-inference.stderr").open("x") as stderr,
    ):
        result = subprocess.run(  # noqa: S603 -- explicit local interpreter/CLI
            command, stdout=stdout, stderr=stderr, check=False, cwd=root
        )
    if result.returncode:
        raise RuntimeError(f"inference returned {result.returncode}; inspect logs")
    scores = load_scores(output / "scores.jsonl")
    by_id = {r.sample_id: r for r in records}
    for score in scores:
        record = by_id[score.sample_id]
        if any(
            getattr(score, field) != getattr(record, field)
            for field in ("label", "generator", "source", "content_type", "modality")
        ):
            raise ValueError("inference score metadata mismatch")
    overall = evaluate_records(scores, expected_sample_ids=list(by_id))
    by_attack = {
        attack: asdict(
            evaluate_records(
                [
                    score
                    for score in scores
                    if score.label == 0 or score.generator == attack
                ]
            )
        )
        for attack in sorted({r.generator for r in records if r.label == 1})
    }
    by_speaker = {
        speaker: asdict(
            evaluate_records(
                [
                    score
                    for score in scores
                    if by_id[score.sample_id].identity_id == speaker
                ]
            )
        )
        for speaker in sorted({r.identity_id for r in records})
    }
    report = {
        "finished_at": utc_now(),
        "script_sha256": file_sha256(Path(__file__)),
        "config_sha256": file_sha256(config_path),
        "scores_sha256": file_sha256(output / "scores.jsonl"),
        "dev_manifest_sha256": file_sha256(dev_path),
        "checkpoint_sha256": checkpoint_hash,
        "overall": asdict(overall),
        "by_attack_versus_all_real": by_attack,
        "by_speaker": by_speaker,
        "attack_diagnostic_note": "Each attack uses the same 256 real dev samples; buckets are not independent.",
        "selection_note": "Dev was used for early stopping; point estimates only, no holdout or formal CI/promotion.",
        "eval_or_holdout_accessed": False,
    }
    atomic_json(experiment / "dev-diagnostics.json", report)
    print(
        json.dumps(
            {
                "run": args.run,
                "eer": overall.fullset_eer,
                "failures": overall.failure_count,
            }
        )
    )


if __name__ == "__main__":
    main()
